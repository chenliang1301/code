cmake_minimum_required(VERSION 3.10)
project(tennis_pose_core_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(tennis_core ../cpp/tennis_analyzer.cc)
target_include_directories(tennis_core PUBLIC ../cpp)
add_executable(tennis_analyzer_test tennis_analyzer_test.cc)
target_link_libraries(tennis_analyzer_test PRIVATE tennis_core)
# These are assertion-based behavioral tests, including in Release builds.
if(MSVC)
    target_compile_options(tennis_analyzer_test PRIVATE /UNDEBUG)
else()
    target_compile_options(tennis_analyzer_test PRIVATE -UNDEBUG)
endif()
add_test(NAME tennis_analyzer COMMAND tennis_analyzer_test)
add_executable(tennis_replay ../tools/tennis_replay.cc)
target_link_libraries(tennis_replay PRIVATE tennis_core)
